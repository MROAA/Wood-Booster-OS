GODFILE:

  id: FILE_SECURITY_ENGINE

  version: 1.0

  description: >
    Tiedostojen, hakemistojen ja

    hallinnan moottori.

  statement:

    >
    Jokainen tiedosto kuuluu tiettyyn
    suojaustasoon ja sitä käsitellään
    tarkoituksen mukaisesti.

  PUBLIC:


    description:

      >
      Tiedosto voidaan jakaa avoimesti.



    examples:


      - documentation
      - public_assets
      - examples





INTERNAL:


    description:

      >
      Järjestelmän sisäinen tiedosto.



    examples:


      - source_code
      - development_notes
      - project_files





CONFIDENTIAL:


    description:

      >
      Tiedosto sisältää liiketoiminta-
      tai käyttäjäkohtaista tietoa.



    examples:


      - customer_files
      - private_documents
      - internal_reports





CRITICAL:


    description:

      >
      Tiedosto jonka paljastuminen
      voi vaarantaa järjestelmän.



    examples:


      - credentials
      - .env_files
      - private_keys
      - security_configuration

  protect:


    - credentials/
    - secrets/
    - private/
    - security/
    - configuration/
    - .env_files

  before_access:


    check:


      - user_permission
      - file_classification
      - operation_type

    require:


      - authorized_access





  WRITE:


    require:


      - permission_check
      - validation
      - backup_if_needed





  DELETE:


    require:


      - confirmation
      - authorization
      - audit_record





  EXECUTE:


    require:


      - trusted_source
      - permission
      - security_check

    - database_schema

    - unauthorized_scripts
