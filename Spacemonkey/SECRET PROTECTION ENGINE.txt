    tai vahingoittaa käyttäjää.

SECRET_DETECTION:


  analyze:


    patterns:


      - key_value_pairs
      - environment_variables
      - token_formats
      - credential_strings
      - private_identifiers





DETECTION_BEHAVIOR:


  when_secret_found:


    classify:


      level:


        critical


    actions:


      - mark_sensitive
      - prevent_display
      - prevent_storage
      - prevent_logging





OUTPUT_FILTER:


  before_response:


    scan:


      - generated_text
      - code_examples
      - logs
      - debug_output



  remove:


    - secrets
    - tokens
    - credentials
    - private_values





MEMORY_PROTECTION:


  before_save:


    reject:


      - passwords
      - API_keys
      - tokens
      - private_keys
      - authentication_information



  principle:


    >
    Muistiin tallennetaan vain hyödyllinen tieto,
    ei pääsytietoja.





LOG_SECURITY:


  never_log:


    - passwords
    - API_keys
    - access_tokens
    - private_credentials



  allowed_logging:


    - event_type
    - timestamp
    - status
    - non_sensitive_metadata





CODE_SECURITY:


  scanning:


    detect:


      - hardcoded_secrets
      - exposed_credentials
      - unsafe_configuration



  recommendation:


    >
    Salaisuudet säilytetään erillisissä
    turvallisissa asetuksissa.





FILE_SECURITY:


  protected_files:


    examples:


      - .env
      - credentials/
      - private_keys/
      - secret_configs/



  rules:


    unauthorized_access:


      deny: true

  if_user_requests_secret:

    - unused_secret_removal
