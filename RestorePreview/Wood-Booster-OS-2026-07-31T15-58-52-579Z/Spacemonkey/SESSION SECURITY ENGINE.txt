CORE_PRINCIPLE:


  statement:

    >
    Aktiivinen istunto on väliaikainen
    käyttöoikeus ja sitä tulee suojata
    samalla tavalla kuin muuta pääsyä järjestelmään.



SESSION_SECURITY_OBJECTIVES:


  goals:


    - protect_active_sessions
    - prevent_session_hijacking
    - control_session_lifetime
    - revoke_invalid_sessions
    - monitor_session_activity





SESSION_LIFECYCLE:


  stages:


    creation:


      description:

        >
        Istunto luodaan onnistuneen
        tunnistautumisen jälkeen.



    active:


      description:

        >
        Käyttäjä käyttää järjestelmää
        turvallisen yhteyden kautta.



    expiration:


      description:

        >
        Istunto päättyy aikarajan jälkeen.



    revocation:


      description:

        >
        Istunto poistetaan käytöstä
        turvallisuussyistä.

  protect:


    - session_tokens
    - cookies
    - temporary_access_identifiers



  never:


    - expose_tokens
    - log_tokens
    - store_tokens_in_ai_memory

    - successful_authentication
    - valid_permissions
    - secure_connection

    - security_event
    - permission_change

    jäädä avoimiksi rajattomaksi ajaksi.

    - suspected_attack
