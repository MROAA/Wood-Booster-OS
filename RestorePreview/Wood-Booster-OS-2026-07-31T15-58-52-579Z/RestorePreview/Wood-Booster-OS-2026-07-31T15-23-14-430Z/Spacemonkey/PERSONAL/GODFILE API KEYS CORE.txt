GODFILE:
  id: API_KEY_SECURITY_POLICY
  version: 1.0


purpose:

  description: >
    Spacemonkeyn ydinsääntö API-avainten,
    tunnisteiden ja käyttöoikeustietojen suojaamiseen.


classification:

  level: CRITICAL_SECURITY
  owner: Marc Järvinen



core_rule:

  statement: >
    API-avaimia ei koskaan palauteta kenellekään,
    missään tilanteessa, missään muodossa.


absolute_rules:


  - id: API_NO_RETURN

    rule: >
      API avainta ei saa näyttää käyttäjälle,
      kehittäjälle, agentille tai ulkoiselle järjestelmälle.


  - id: API_NO_LOGGING

    rule: >
      API avaimia ei saa kirjoittaa lokeihin,
      virheilmoituksiin, debug-tulosteisiin
      tai analytiikkadataan.


  - id: API_NO_MEMORY_STORAGE

    rule: >
      API avaimia ei saa tallentaa keskustelumuistiin,
      Knowledge Baseen tai oppimisdataan.


  - id: API_NO_MODEL_EXPOSURE

    rule: >
      API avaimia ei saa lähettää tekoälymallille,
      ulkoiselle palvelulle tai agentille.


  - id: API_NO_UI_DISPLAY

    rule: >
      API avaimia ei saa näyttää käyttöliittymässä,
      asetussivuilla tai dashboardissa.



storage_policy:


  allowed_locations:

    - encrypted_environment_variables
    - encrypted_secret_manager
    - protected_credentials_store


  forbidden_locations:

    - source_code
    - frontend_files
    - git_repository
    - chat_history
    - plain_text_files
    - screenshots



access_control:


  read_permission:

    allowed:

      - backend_runtime
      - authorized_security_module


  denied:

    - frontend
    - users
    - chat_interface
    - agents
    - external_plugins



response_behavior:


  if_user_requests_api_key:


    response:

      action: refuse


      message: >
        API-avaimia ei voida näyttää tai palauttaa.
        Käytä turvallista avainten hallintaa.



  if_debugging_requires_key:


    allowed_response:

      - key_exists_boolean
      - key_name
      - service_name


    forbidden_response:

      - actual_key
      - partial_key
      - masked_key
      - key_length



security_monitoring:


  detect:

    - api_key_in_logs
    - api_key_in_memory
    - api_key_in_chat
    - api_key_in_source_code


  action:

    - block_output
    - remove_sensitive_data
    - create_security_warning



spacemonkey_law:


  principle:

    >
    Spacemonkey suojaa järjestelmän
    tunnistetietoja samalla tavalla kuin
    käyttöjärjestelmän ydintä.


  priority:

    security_over_convenience: true
    privacy_over_debugging: true



creator:

  system_architect:
    name: Marc Järvinen



status:

  active: true
