GODFILE:

  id: INCIDENT_RESPONSE_ENGINE

  version: 1.0


purpose:

  description: >
    Tietoturvatapahtumien tunnistamisen,
    käsittelyn, rajaamisen ja palautumisen
    turvallisuusmoottori.



CORE_PRINCIPLE:


  statement:

    >
    Turvallisuusongelmaan reagoidaan
    hallitusti, nopeasti ja dokumentoidusti.



INCIDENT_LIFECYCLE:


  phases:


    detection:


      description:

        >
        Tunnistetaan mahdollinen turvallisuusongelma.



      actions:


        - collect_safe_information
        - classify_event
        - determine_severity





    analysis:


      description:

        >
        Selvitetään tapahtuman laajuus
        ilman tarpeetonta tiedon keräämistä.



      actions:


        - identify_affected_component
        - evaluate_risk
        - determine_response





    containment:


      description:

        >
        Rajataan ongelman vaikutus.



      actions:


        - isolate_component
        - restrict_access
        - stop_unsafe_operations





    recovery:


      description:

        >
        Palautetaan turvallinen toimintatila.



      actions:


        - verify_system_state
        - restore_safe_configuration
        - monitor_recovery





    review:


      description:

        >
        Opitaan tapahtumasta.



      actions:


        - document_lessons
        - improve_controls
        - update_security_rules

  authentication_incident:


    examples:


      - repeated_failed_logins
      - unauthorized_access_attempt





  data_incident:


    examples:


      - accidental_exposure
      - incorrect_access





  software_incident:


    examples:


      - vulnerable_dependency
      - malicious_component





  system_incident:


    examples:


      - unexpected_behavior
      - service_failure





  AI_SECURITY_INCIDENT:


    examples:


      - prompt_injection_attempt
      - unauthorized_instruction
      - unsafe_output





SEVERITY_LEVELS:


  LOW:


    description:

      >
      Pieni vaikutus, ei välitöntä vaaraa.





  MEDIUM:


    description:

      >
      Vaatii tarkistusta ja korjaavia toimia.





  HIGH:


    description:

      >
      Voi vaikuttaa järjestelmän turvallisuuteen.





  CRITICAL:


    description:

      >
      Vakava vaikutus järjestelmän tai
      tietojen turvallisuuteen.

  always:


    - remain_calm
    - collect_only_needed_information
    - protect_secrets
    - prioritize_user_safety

      - expose_tokens
      - expose_private_keys
