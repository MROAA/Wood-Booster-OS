    Zero Trust -periaatteen mukainen

CORE_PRINCIPLE:


  statement:

    >
    Älä koskaan luota automaattisesti.
    Varmista aina käyttäjä,
    komponentti, toiminto ja resurssi.



ZERO_TRUST_MODEL:


  principles:


    verify_every_request:


      description:

        >
        Jokainen toiminto tarkistetaan
        ennen suorittamista.





    least_privilege:


      description:

        >
        Käyttöoikeudet annetaan vain
        tarvittavaan tarkoitukseen.





    assume_breach:


      description:

        >
        Järjestelmä toimii oletuksella,
        että jokin osa voi vaarantua.





IDENTITY_VERIFICATION:


  verify:


    - user_identity
    - component_identity
    - service_identity
    - agent_identity





ACCESS_DECISION:


  before_access:


    evaluate:


      - who_requests
      - what_action
      - what_resource
      - why_needed
      - current_security_state





TRUST_LEVELS:


  UNKNOWN:


    action:


      >
      Ei pääsyä ennen varmennusta.





  VERIFIED:


    action:


      >
      Sallittu vain määritellyillä oikeuksilla.





  PRIVILEGED:


    action:


      >
      Vaatii lisävahvistuksen ja auditoinnin.
