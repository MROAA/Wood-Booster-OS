GODFILE:

  id: PERMISSION_ENGINE

  version: 1.0


purpose:

  description: >
    Toimintojen, resurssien ja AI-komponenttien
    yksityiskohtaisen käyttöoikeuden hallintamoottori.



CORE_PRINCIPLE:


  statement:

    >
    Jokainen toiminto arvioidaan ennen suorittamista.
    Oikeus täytyy olla olemassa ennen toimintaa.





PERMISSION_MODEL:


  permission_check:


    steps:


      1:

        identify_actor


      2:

        identify_action


      3:

        identify_resource


      4:

        check_permission


      5:

        allow_or_deny





PERMISSION_TYPES:


  READ:


    description:


      >
      Oikeus lukea tietoa.



    examples:


      - read_document
      - view_project
      - query_knowledge





  WRITE:


    description:


      >
      Oikeus muuttaa tai tallentaa tietoa.



    examples:


      - update_file
      - save_memory
      - modify_record





  EXECUTE:


    description:


      >
      Oikeus suorittaa toiminto.



    examples:


      - run_command
      - start_service
      - execute_workflow





  DELETE:


    description:


      >
      Oikeus poistaa tietoa.



    restrictions:


      - confirmation_required
      - audit_required





  ADMIN:


    description:


      >
      Järjestelmän hallintaoikeus.



    restrictions:


      - limited_users
      - security_logging
      - explicit_authorization





AI_AGENT_PERMISSIONS:


  principle:


    >
    Jokaisella agentilla on oma rajattu tehtäväalue.





  examples:


    workshop_agent:


      allowed:


        - analyze_materials
        - explain_workflow
        - answer_workshop_questions



      denied:


        - modify_security
        - access_credentials





  marketing_agent:


      allowed:


        - create_text
        - analyze_brand_language



      denied:


        - access_private_data
        - modify_system_core





  coding_agent:


      allowed:


        - analyze_code
        - suggest_changes
        - create_files



      denied:


        - expose_secrets
        - bypass_security





TOOL_PERMISSIONS:


  every_tool_requires:


    - defined_purpose
    - allowed_permission
    - security_check





FILE_PERMISSIONS:


  protected_files:


    critical:


      - security_rules
      - credentials
      - environment_files
      - private_configuration



  access:


    require:


      - authorization
      - logging
      - validation





ACTION_DECISION:


  before_action:


    ask:


      - Who is requesting?
      - What action?
      - Which resource?
      - Is permission valid?





DENIAL_BEHAVIOR:


  when_denied:


    response:


      >
      Toimintoa ei suoritettu,
      koska tarvittava käyttöoikeus puuttuu.



  never:


    - bypass_permission
    - silently_escalate
    - ignore_security





PERMISSION_CHANGES:


  rules:


    changes_require:


      - authorization
      - logging
      - review





AUDIT:


  track:


    - permission_checks
    - denied_actions
    - granted_actions
    - changes





MEMORY_SECURITY:


  never_store:


    - permission_secrets
    - access_tokens
    - private_credentials
