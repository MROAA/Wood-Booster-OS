DATA_LEVELS:


  PUBLIC:


    description:

      >
      Tieto, jonka voi julkaista
      ilman turvallisuusriskiä.



    examples:


      - website_content
      - public_documentation
      - marketing_material
      - product_information



    handling:


      allowed:


        - share
        - publish
        - index





  INTERNAL:


    description:

      >
      Järjestelmän sisäinen tieto,
      jota ei ole tarkoitettu ulkopuolisille.



    examples:


      - architecture_notes
      - development_plans
      - internal_processes
      - technical_documentation



    handling:


      allowed:


        - internal_use
        - controlled_access



      forbidden:


        - public_release_without_review





  CONFIDENTIAL:


    description:

      >
      Tieto jonka vuotaminen voi aiheuttaa haittaa.



    examples:


      - customer_information
      - business_plans
      - private_documents
      - internal_credentials_metadata



    handling:


      required:


        - access_control
        - limited_storage
        - audit





  CRITICAL:


    description:

      >
      Tieto jonka paljastuminen voi
      vaarantaa järjestelmän tai käyttäjän.



    examples:


      - passwords
      - API_keys
      - private_keys
      - authentication_secrets



    handling:


      rules:


        - never_display
        - never_store_in_ai_memory
        - never_log
        - restricted_access
